/* ============================================================
   UTILITIES

const $ = s => document.querySelector(s);
const term = $('#term');
const input = $('#cmdInput');

function esc(s){
  return String(s).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
}
function print(html){
  const d = document.createElement('div');
  d.className = 'line';
  d.innerHTML = html;
  term.appendChild(d);
  term.scrollTop = term.scrollHeight;
}
function out(t){ print(esc(t)); }
function ok(t){ print('<span class="green">'+esc(t)+'</span>'); }
function err(t){ print('<span class="red">'+esc(t)+'</span>'); }
function dim(t){ print('<span class="dim">'+esc(t)+'</span>'); }
function blank(){ print(''); }

function tokenize(str){
  const outArr = []; let cur = ''; let q = null;
  for (let i=0;i<str.length;i++){
    const ch = str[i];
    if (q){ if (ch === q) q = null; else cur += ch; }
    else if (ch === '"' || ch === "'") q = ch;
    else if (/\s/.test(ch)){ if (cur){ outArr.push(cur); cur = ''; } }
    else cur += ch;
  }
  if (cur) outArr.push(cur);
  return outArr;
}

function newSha(){
  const hex = '0123456789abcdef';
  let s = '';
  for (let i=0;i<40;i++) s += hex[Math.floor(Math.random()*16)];
  return s;
}
const short = s => (s||'').slice(0,7);

function fmtDate(ts){
  const d = new Date(ts);
  const p = n => String(n).padStart(2,'0');
  return `${d.getFullYear()}-${p(d.getMonth()+1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
}

/* LCS line diff -> array of [' '|'+'|'-', line] */
function lcsDiff(a, b){
  const n = a.length, m = b.length;
  const dp = Array.from({length:n+1}, () => new Uint32Array(m+1));
  for (let i=n-1;i>=0;i--)
    for (let j=m-1;j>=0;j--)
      dp[i][j] = a[i] === b[j] ? dp[i+1][j+1]+1 : Math.max(dp[i+1][j], dp[i][j+1]);
  const ops = []; let i=0, j=0;
  while (i<n && j<m){
    if (a[i] === b[j]){ ops.push([' ', a[i]]); i++; j++; }
    else if (dp[i+1][j] >= dp[i][j+1]){ ops.push(['-', a[i]]); i++; }
    else { ops.push(['+', b[j]]); j++; }
  }
  while (i<n) ops.push(['-', a[i++]]);
  while (j<m) ops.push(['+', b[j++]]);
  return ops;
}
const toLines = c => (c === undefined || c === '') ? [] : String(c).replace(/\n$/, '').split('\n');

/* ============================================================
   STATE
   ============================================================ */
let CLOCK = 1700000000000;
const tick = () => (CLOCK += 60000);

function blankState(){
  return {
    init:false,
    config:{ name:'', email:'' },
    head:'main',          // string -> branch name ; {sha} -> detached
    branches:{},          // name -> sha
    commits:{},           // sha -> {sha,msg,parents,tree,author,email,time}
    work:{},              // filename -> content
    index:{},             // filename -> content
    remotes:{},           // name -> url
    remoteRefs:{},        // name -> {branch: sha}
    stash:[],
    reflog:[],
    mergeState:null
  };
}
let S = blankState();

/* ---- derived helpers ---- */
function headSha(){
  if (typeof S.head === 'string') return S.branches[S.head] || null;
  return (S.head && S.head.sha) || null;
}
function headTree(){
  const s = headSha();
  return s && S.commits[s] ? S.commits[s].tree : {};
}
function branchLabel(){
  return typeof S.head === 'string' ? S.head : 'HEAD detached at ' + short(S.head.sha);
}
function promptStr(){
  if (!S.init) return 'user@git:~/project$';
  return `user@git:~/project (${branchLabel()})$`;
}
function authorName(){ return S.config.name || 'You'; }
function authorEmail(){ return S.config.email || 'you@example.com'; }

function newCommit(tree, msg, parents){
  const sha = newSha();
  S.commits[sha] = {
    sha, msg, parents: parents.slice(),
    tree: Object.assign({}, tree),
    author: authorName(), email: authorEmail(), time: tick()
  };
  return sha;
}

function ancestorsOf(sha){
  const set = new Set(); const st = [sha];
  while (st.length){
    const s = st.pop();
    if (!s || set.has(s)) continue;
    set.add(s);
    const c = S.commits[s];
    if (c) st.push(...c.parents);
  }
  return set;
}
function isAncestor(a, b){
  if (!a || !b) return false;
  return ancestorsOf(b).has(a);
}
function mergeBase(a, b){
  if (!a || !b) return null;
  const anc = ancestorsOf(a);
  const q = [b], seen = new Set();
  while (q.length){
    const s = q.shift();
    if (seen.has(s)) continue;
    seen.add(s);
    if (anc.has(s)) return s;
    const c = S.commits[s];
    if (c) q.push(...c.parents);
  }
  return null;
}

function resolveRev(ref){
  if (!ref) return null;
  if (ref === 'HEAD' || ref === '@') return headSha();
  let m = ref.match(/^HEAD~(\d+)$/);
  if (m){
    let s = headSha(), n = +m[1];
    while (n-- > 0 && s){ const c = S.commits[s]; s = c ? c.parents[0] : null; }
    return s || null;
  }
  if (ref === 'HEAD^'){
    const s = headSha(); const c = s && S.commits[s];
    return c ? (c.parents[0] || null) : null;
  }
  if (Object.prototype.hasOwnProperty.call(S.branches, ref)) return S.branches[ref];
  if (ref.includes('/')){
    const idx = ref.indexOf('/');
    const r = ref.slice(0, idx), b = ref.slice(idx+1);
    if (S.remoteRefs[r] && S.remoteRefs[r][b]) return S.remoteRefs[r][b];
  }
  const found = Object.keys(S.commits).find(s => s.startsWith(ref));
  return found || null;
}

function checkoutTree(tree){
  S.work = Object.assign({}, tree);
  S.index = Object.assign({}, tree);
}

function reflog(msg){
  S.reflog.push({ t: fmtDate(tick()), from: branchLabel(), msg });
}

/* ---- status analysis ---- */
function analyze(){
  const head = headTree();
  const all = new Set([...Object.keys(S.work), ...Object.keys(S.index), ...Object.keys(head)]);
  const res = {};
  for (const f of all){
    const inW = f in S.work, inI = f in S.index, inH = f in head;
    const r = { file:f, staged:null, unstaged:null, untracked:false };
    if (inI && !inH) r.staged = 'new file';
    else if (!inI && inH) r.staged = 'deleted';
    else if (inI && inH && S.index[f] !== head[f]) r.staged = 'modified';

    if (inW && !inI && !inH) r.untracked = true;
    else if (inW && inI && S.work[f] !== S.index[f]) r.unstaged = 'modified';
    else if (!inW && inI) r.unstaged = 'deleted';
    res[f] = r;
  }
  return res;
}

/* ============================================================
   RENDERING
   ============================================================ */
function render(){
  const st = analyze();

  /* ---- working directory ---- */
  const wl = $('#workList');
  const workFiles = [...new Set([...Object.keys(S.work), ...Object.keys(headTree())])].sort();
  $('#workCount').textContent = workFiles.length;
  if (!workFiles.length){
    wl.innerHTML = '<div class="empty">No files yet — use the ＋ button or create one below.</div>';
  } else {
    wl.innerHTML = '';
    for (const f of workFiles){
      const r = st[f] || {};
      let badge = '', cls = '';
      if (r.untracked){ badge = 'U'; cls = 'b-unt'; }
      else if (r.unstaged === 'modified'){ badge = 'M'; cls = 'b-mod'; }
      else if (r.unstaged === 'deleted'){ badge = 'D'; cls = 'b-del'; }
      else if (r.staged === 'new file'){ badge = 'A'; cls = 'b-new'; }
      else if (r.staged === 'modified'){ badge = 'M'; cls = 'b-stg'; }
      else if (r.staged === 'deleted'){ badge = 'D'; cls = 'b-del'; }

      const row = document.createElement('div');
      row.className = 'frow';
      row.innerHTML = `<span style="opacity:.6">📄</span>
        <span class="fname">${esc(f)}</span>
        ${badge ? `<span class="badge ${cls}">${badge}</span>` : ''}`;
      row.onclick = () => openEditor(f);
      wl.appendChild(row);
    }
  }
  const addBtn = document.createElement('div');
  addBtn.className = 'frow';
  addBtn.style.color = 'var(--dim)';
  addBtn.innerHTML = '<span>＋</span><span class="fname">new file…</span>';
  addBtn.onclick = () => {
    const name = prompt('New file name:', 'notes.txt');
    if (!name) return;
    S.work[name] = '';
    render();
    openEditor(name);
  };
  wl.appendChild(addBtn);

  /* ---- staging area ---- */
  const sl = $('#stageList');
  const stagedFiles = Object.keys(S.index).filter(f => {
    const inH = f in headTree();
    return !inH || S.index[f] !== headTree()[f];
  });
  $('#stageCount').textContent = stagedFiles.length;
  if (!stagedFiles.length){
    sl.innerHTML = '<div class="empty">Nothing staged — run <code style="color:var(--green)">git add &lt;file&gt;</code></div>';
  } else {
    sl.innerHTML = '';
    for (const f of stagedFiles.sort()){
      const inH = f in headTree();
      const type = !inH ? 'new file' : 'modified';
      const row = document.createElement('div');
      row.className = 'frow';
      row.innerHTML = `<span style="opacity:.6">✔</span>
        <span class="fname">${esc(f)}</span>
        <span class="badge ${type === 'new file' ? 'b-new' : 'b-stg'}">${type === 'new file' ? 'A' : 'M'}</span>`;
      sl.appendChild(row);
    }
  }

  /* ---- branches ---- */
  const bl = $('#branchList');
  const localNames = Object.keys(S.branches).sort();
  const remoteNames = Object.entries(S.remoteRefs).flatMap(([r, refs]) => Object.keys(refs).map(b => r + '/' + b));
  if (!localNames.length && !remoteNames.length){
    bl.innerHTML = '<div class="empty">No branches yet</div>';
  } else {
    bl.innerHTML = '';
    for (const b of localNames){
      const isCur = typeof S.head === 'string' && S.head === b;
      const row = document.createElement('div');
      row.className = 'brow' + (isCur ? ' current' : '');
      row.innerHTML = `<span class="starb">${isCur ? '★' : ' '}</span>
        <span class="bname">${esc(b)}</span>
        <span style="color:var(--yellow);font-size:11px">${short(S.branches[b])}</span>`;
      bl.appendChild(row);
    }
    for (const r of remoteNames){
      const row = document.createElement('div');
      row.className = 'brow';
      row.innerHTML = `<span style="opacity:.5">☁</span>
        <span class="bname remote">${esc(r)}</span>
        <span style="color:var(--yellow);font-size:11px">${short(S.remoteRefs[r.split('/')[0]][r.split('/').slice(1).join('/')])}</span>`;
      bl.appendChild(row);
    }
    if (typeof S.head !== 'string'){
      const row = document.createElement('div');
      row.className = 'brow current';
      row.innerHTML = `<span style="color:var(--purple)">◉</span><span class="bname">HEAD (detached)</span>`;
      bl.appendChild(row);
    }
  }

  /* ---- commit graph ---- */
  const g = $('#graph');
  const shas = Object.keys(S.commits).sort((a,b) => S.commits[b].time - S.commits[a].time);
  if (!shas.length){
    g.innerHTML = '<div class="empty">No commits yet</div>';
  } else {
    const labels = {};
    const push = (sha, text, cls) => {
      if (!sha) return;
      (labels[sha] = labels[sha] || []).push({text, cls});
    };
    for (const [name, sha] of Object.entries(S.branches)){
      const isCur = typeof S.head === 'string' && S.head === name;
      push(sha, (isCur ? 'HEAD → ' : '') + name, isCur ? 'head' : 'branch');
    }
    for (const [r, refs] of Object.entries(S.remoteRefs))
      for (const [b, sha] of Object.entries(refs)) push(sha, r + '/' + b, 'remote');
    if (typeof S.head !== 'string') push(S.head.sha, 'HEAD', 'head');

    g.innerHTML = '';
    for (const sha of shas){
      const c = S.commits[sha];
      const lbls = (labels[sha] || []).map(l => `<span class="lbl ${l.cls}">${esc(l.text)}</span>`).join('');
      const row = document.createElement('div');
      row.className = 'crow';
      row.innerHTML = `
        <div class="cgutter"><span class="cdot ${c.parents.length > 1 ? 'merge' : ''}"></span></div>
        <div class="csha">${short(sha)}</div>
        <div class="cmsg">${esc(c.msg.split('\n')[0])}${lbls}</div>`;
      row.title = `${c.author} — ${fmtDate(c.time)}\n${c.msg}`;
      g.appendChild(row);
    }
  }

  /* ---- prompt ---- */
  $('#promptLabel').textContent = promptStr();
}

/* ============================================================
   COMMAND IMPLEMENTATION
   ============================================================ */
function exec(a){
  if (!a.length){ printHelp(); return; }
  const sub = a[0], args = a.slice(1);

  // commands allowed outside a repo
  switch (sub){
    case 'init':   return cmdInit(args);
    case 'clone':  return cmdClone(args);
    case 'config': return cmdConfig(args);
    case 'help':   return printHelp();
    case '--version': return out('git version 2.43.0');
  }

  if (!S.init){
    err('fatal: not a git repository (or any of the parent directories): .git');
    dim('hint: run "git init" or "git clone <url>" first');
    return;
  }

  switch (sub){
    case 'add':        return cmdAdd(args);
    case 'status':     return cmdStatus();
    case 'diff':       return cmdDiff(args);
    case 'mv':         return cmdMv(args);
    case 'rm':         return cmdRm(args);
    case 'commit':     return cmdCommit(args);
    case 'branch':     return cmdBranch(args);
    case 'switch':     return cmdSwitch(args);
    case 'checkout':   return cmdCheckout(args);
    case 'merge':      return cmdMerge(args);
    case 'rebase':     return cmdRebase(args);
    case 'cherry-pick':return cmdCherryPick(args);
    case 'remote':     return cmdRemote(args);
    case 'fetch':      return cmdFetch(args);
    case 'pull':       return cmdPull(args);
    case 'push':       return cmdPush(args);
    case 'restore':    return cmdRestore(args);
    case 'reset':      return cmdReset(args);
    case 'revert':     return cmdRevert(args);
    case 'clean':      return cmdClean(args);
    case 'stash':      return cmdStash(args);
    case 'log':        return cmdLog(args);
    case 'show':       return cmdShow(args);
    case 'blame':      return cmdBlame(args);
    case 'reflog':     return cmdReflog();
    default:
      err(`git: '${sub}' is not a git command. See 'git help'.`);
  }
}

function cmdConfig(args){
  const rest = [];
  let scope = 'local';
  for (const x of args){
    if (x === '--global' || x === '-g') scope = 'global';
    else if (x === '--local') scope = 'local';
    else rest.push(x);
  }
  if (!rest.length){
    out('user.name=' + (S.config.name || ''));
    out('user.email=' + (S.config.email || ''));
    return;
  }
  const key = rest[0];
  const prop = key.split('.').pop();
  if (rest.length === 1){ out(S.config[prop] || ''); return; }
  S.config[prop] = rest[1];
  ok(`config set: ${key} = ${rest[1]} (${scope})`);
}

/* ---------- staging ---------- */
function cmdAdd(args){
  const targets = args.filter(x => !x.startsWith('-'));
  const all = args.includes('-A') || args.includes('--all') || args.includes('-u');
  if (!targets.length && !all){
    err('Nothing specified, nothing added.');
    dim('hint: "git add ." or "git add <file>"');
    return;
  }
  const head = headTree();
  const addAll = all || targets.includes('.') || targets.includes('*') || targets.includes('-A');

  if (addAll){
    for (const f of Object.keys(S.work)) S.index[f] = S.work[f];
    for (const f of Object.keys(S.index)) if (!(f in S.work)) delete S.index[f];
    ok('staged all changes (including deletions)');
    return;
  }
  for (const t of targets){
    if (t in S.work){
      S.index[t] = S.work[t];
      ok(`add '${t}'`);
    } else if (t in S.index || t in head){
      delete S.index[t];
      ok(`rm '${t}'`);
    } else {
      err(`fatal: pathspec '${t}' did not match any files`);
    }
  }
}

function cmdStatus(){
  const head = headTree();
  const st = analyze();
  print(`On branch <span class="cyan">${esc(branchLabel())}</span>`);
  const rem = S.remoteRefs.origin && S.remoteRefs.origin[typeof S.head === 'string' ? S.head : 'main'];
  if (rem && typeof S.head === 'string'){
    const localSha = S.branches[S.head];
    if (rem === localSha) out(`Your branch is up to date with 'origin/${S.head}'.`);
    else if (isAncestor(rem, localSha)) out(`Your branch is ahead of 'origin/${S.head}' by ${countAhead(rem, localSha)} commit(s).`);
    else if (isAncestor(localSha, rem)) out(`Your branch is behind 'origin/${S.head}' by ${countAhead(localSha, rem)} commit(s).`);
    else out(`Your branch and 'origin/${S.head}' have diverged.`);
  }
  if (!headSha()) out('No commits yet');
  blank();

  const staged = [], unstaged = [], untracked = [];
  for (const f of Object.keys(st)) {
    const r = st[f];
    if (r.staged) staged.push(r);
    if (r.unstaged) unstaged.push(r);
    if (r.untracked) untracked.push(r);
  }

  if (staged.length){
    out('Changes to be committed:');
    dim('  (use "git restore --staged <file>..." to unstage)');
    for (const r of staged) print(`        <span class="green">${r.staged}:</span>   ${esc(r.file)}`);
    blank();
  }
  if (unstaged.length){
    out('Changes not staged for commit:');
    dim('  (use "git add <file>..." to update what will be committed)');
    dim('  (use "git restore <file>..." to discard changes in working directory)');
    for (const r of unstaged) print(`        <span class="red">${r.unstaged}:</span>   ${esc(r.file)}`);
    blank();
  }
  if (untracked.length){
    out('Untracked files:');
    dim('  (use "git add <file>..." to include in what will be committed)');
    for (const r of untracked) print(`        <span class="red">${esc(r.file)}</span>`);
    blank();
  }
  if (!staged.length && !unstaged.length && !untracked.length){
    out('nothing to commit, working tree clean');
  }
}
function countAhead(from, to){
  const anc = ancestorsOf(from);
  let n = 0;
  for (const s of ancestorsOf(to)) if (!anc.has(s)) n++;
  return n;
}

function printDiffHeader(name){
  print(`<span class="dim">diff --git a/${esc(name)} b/${esc(name)}</span>`);
  print(`<span class="dim">--- a/${esc(name)}</span>`);
  print(`<span class="dim">+++ b/${esc(name)}</span>`);
}
function printDiffBody(oldC, newC){
  const a = toLines(oldC), b = toLines(newC);
  print(`<span class="cyan">@@ -1,${a.length} +1,${b.length} @@</span>`);
  for (const [op, line] of lcsDiff(a, b)){
    if (op === '+') print('<span class="green">+' + esc(line) + '</span>');
    else if (op === '-') print('<span class="red">-' + esc(line) + '</span>');
    else print('<span class="dim"> ' + esc(line) + '</span>');
  }
}

function cmdDiff(args){
  const head = headTree();
  let any = false;
  const files = new Set([...Object.keys(S.work), ...Object.keys(S.index)]);
  for (const f of [...files].sort()){
    const inW = f in S.work, inI = f in S.index, inH = f in head;
    if (inW && !inI && !inH) continue;              // untracked — git diff ignores
    const i = inI ? S.index[f] : '';
    const w = inW ? S.work[f] : '';
    if (i === w) continue;
    any = true;
    printDiffHeader(f);
    printDiffBody(i, w);
  }
  if (!any) dim('(no unstaged changes)');
}

function cmdMv(args){
  const [o, n] = args;
  if (!o || !n){ err('usage: git mv <old> <new>'); return; }
  if (!(o in S.work)){ err(`fatal: bad source, source=${o}, destination=${n}`); return; }
  S.work[n] = S.work[o]; delete S.work[o];
  if (o in S.index){ S.index[n] = S.index[o]; delete S.index[o]; }
  ok(`renamed: ${o} -> ${n}`);
}

function cmdRm(args){
  const files = args.filter(x => !x.startsWith('-'));
  if (!files.length){ err('fatal: No pathspec was given.'); return; }
  for (const f of files){
    if (!(f in S.work) && !(f in S.index)){ err(`fatal: pathspec '${f}' did not match any files`); continue; }
    delete S.work[f];
    delete S.index[f];
    ok(`rm '${f}'`);
  }
}

/* ---------- commit ---------- */
function cmdCommit(args){
  let amend = false, msg = null, stageAll = false;
  for (let i=0;i<args.length;i++){
    const x = args[i];
    if (x === '--amend') amend = true;
    else if (x === '-a' || x === '--all') stageAll = true;
    else if (x === '-am' || x === '-ma'){ stageAll = true; amend = false; }
    else if (x === '-m' || x === '--message'){ msg = args[i+1] || ''; i++; }
    else if (x.startsWith('-m') && x.length > 2) msg = x.slice(2);
  }
  if (stageAll){
    const head = headTree();
    for (const f of Object.keys(S.work)) if (f in head) S.index[f] = S.work[f];
  }

  const head = headSha();
  const headT = headTree();
  const changed = [...new Set([...Object.keys(S.index), ...Object.keys(headT)])]
    .filter(f => (f in S.index) !== (f in headT) || S.index[f] !== headT[f]);

  if (!changed.length && !amend && !S.mergeState){
    print(`On branch <span class="cyan">${esc(branchLabel())}</span>`);
    const untracked = Object.keys(S.work).filter(f => !(f in S.index) && !(f in headT));
    if (untracked.length){
      out('nothing added to commit but untracked files present');
      dim('  (use "git add <file>..." to include in what will be committed)');
      for (const f of untracked) print(`        <span class="red">${esc(f)}</span>`);
    } else {
      out('nothing to commit, working tree clean');
    }
    return;
  }
  if (msg === null && !amend){
    err('Aborting commit due to empty commit message.');
    dim('hint: use git commit -m "your message"');
    return;
  }

  let parents;
  if (S.mergeState) parents = S.mergeState.parents.slice();
  else if (amend && head) parents = S.commits[head].parents.slice();
  else parents = head ? [head] : [];

  if (amend && !head){ err('fatal: You have nothing to amend.'); return; }
  if (msg === null && amend) msg = S.commits[head].msg;

  const sha = newCommit(S.index, msg, parents);

  if (typeof S.head === 'string') S.branches[S.head] = sha;
  else S.head = { sha };

  S.mergeState = null;
  reflog(`${amend ? 'commit (amend)' : 'commit'}: ${msg.split('\n')[0]}`);

  const [bname] = [typeof S.head === 'string' ? S.head : 'HEAD detached'];
  const root = parents.length === 0 ? ' (root-commit)' : '';
  print(`<span class="yellow">[${esc(bname)} ${short(sha)}${root}]</span> ${esc(msg.split('\n')[0])}`);
  out(` ${changed.length} file(s) changed`);
}

/* ---------- branching ---------- */
function cmdBranch(args){
  if (!args.length || args[0] === '-a' || args[0] === '-v'){
    const names = Object.keys(S.branches).sort();
    if (!names.length) dim('(no branches yet)');
    for (const b of names){
      const cur = typeof S.head === 'string' && S.head === b;
      print(`${cur ? '<span class="green">*</span>' : ' '} <span class="${cur ? 'green' : ''}">${esc(b)}</span> <span class="dim">${short(S.branches[b])}</span>`);
    }
    if (args[0] === '-a'){
      for (const [r, refs] of Object.entries(S.remoteRefs))
        for (const b of Object.keys(refs))
          print(`  <span class="purple">remotes/${esc(r)}/${esc(b)}</span> <span class="dim">${short(refs[b])}</span>`);
    }
    return;
  }
  if (args[0] === '-d' || args[0] === '-D'){
    const name = args[1];
    if (!name){ err('fatal: branch name required'); return; }
    if (!(name in S.branches)){ err(`error: branch '${name}' not found.`); return; }
    if (typeof S.head === 'string' && S.head === name){
      err(`error: Cannot delete branch '${name}' checked out at '/home/user/project'`);
      return;
    }
    if (args[0] === '-d' && !isAncestor(S.branches[name], headSha())){
      err(`error: The branch '${name}' is not fully merged.`);
      dim(`hint: use "git branch -D ${name}" to force delete`);
      return;
    }
    delete S.branches[name];
    ok(`Deleted branch ${name} (was ${short(S.branches[name])}).`);
    return;
  }
  const name = args[0];
  if (name in S.branches){ err(`fatal: a branch named '${name}' already exists`); return; }
  S.branches[name] = headSha();
  ok(`branch '${name}' created at ${short(headSha()) || '(no commits yet)'}`);
}

function dirtyConflict(targetTree){
  const cur = headTree();
  const dirty = [];
  const files = new Set([...Object.keys(S.work), ...Object.keys(S.index), ...Object.keys(cur), ...Object.keys(targetTree)]);
  for (const f of files){
    const w = S.work[f], i = S.index[f], c = cur[f], t = targetTree[f];
    const localDirty = (w !== i) || (i !== c);
    if (localDirty && c !== t) dirty.push(f);
  }
  return dirty;
}

function switchTo(name, create, args){
  if (create){
    if (name in S.branches){ err(`fatal: a branch named '${name}' already exists`); return; }
    S.branches[name] = headSha();
    ok(`Switched to a new branch '${name}'`);
  } else {
    if (!(name in S.branches)){ err(`fatal: invalid reference: ${name}`); return; }
  }
  const target = S.branches[name];
  const targetTree = target && S.commits[target] ? S.commits[target].tree : {};
  const bad = dirtyConflict(targetTree);
  if (bad.length){
    err('error: Your local changes to the following files would be overwritten by checkout:');
    for (const f of bad) print(`        ${esc(f)}`);
    dim('Please commit your changes or stash them before you switch branches.');
    return;
  }
  const from = branchLabel();
  S.head = name;
  checkoutTree(targetTree);
  reflog(`checkout: moving from ${from} to ${name}`);
  if (!create) ok(`Switched to branch '${name}'`);
}

function cmdSwitch(args){
  if (!args.length){ err('fatal: missing branch name'); return; }
  if (args[0] === '-c' || args[0] === '-C'){
    if (!args[1]){ err('fatal: missing branch name'); return; }
    return switchTo(args[1], true, args);
  }
  switchTo(args[0], false, args);
}

function cmdCheckout(args){
  if (!args.length){ err('fatal: missing branch name'); return; }
  // git checkout -- <file>   /   git checkout <file>
  if (args[0] === '--'){
    const files = args.slice(1);
    if (!files.length){ err('fatal: no pathspec given'); return; }
    for (const f of files){
      if (f in S.index) S.work[f] = S.index[f];
      else delete S.work[f];
      ok(`restored '${f}' from index`);
    }
    return;
  }
  if (args.includes('--')){
    const idx = args.indexOf('--');
    const files = args.slice(idx+1);
    const src = resolveRev(args.slice(0, idx)[0]);
    const tree = src && S.commits[src] ? S.commits[src].tree : {};
    for (const f of files){
      if (f in tree) S.work[f] = tree[f];
      else delete S.work[f];
      ok(`checked out '${f}'`);
    }
    return;
  }
  if (args[0] === '-b' || args[0] === '-B'){
    if (!args[1]){ err('fatal: missing branch name'); return; }
    return switchTo(args[1], true, args);
  }
  const ref = args[0];
  if (ref in S.branches) return switchTo(ref, false, args);
  const sha = resolveRev(ref);
  if (sha){
    const tree = S.commits[sha].tree;
    const bad = dirtyConflict(tree);
    if (bad.length){
      err('error: Your local changes would be overwritten by checkout:');
      for (const f of bad) print(`        ${esc(f)}`);
      return;
    }
    const from = branchLabel();
    S.head = { sha };
    checkoutTree(tree);
    reflog(`checkout: moving from ${from} to ${short(sha)}`);
    dim(`Note: switching to '${short(sha)}'. You are in 'detached HEAD' state.`);
    return;
  }
  err(`error: pathspec '${ref}' did not match any file(s) known to git`);
}

/* ---------- merging ---------- */
function threeWayMerge(baseT, oursT, theirsT){
  const files = new Set([...Object.keys(baseT), ...Object.keys(oursT), ...Object.keys(theirsT)]);
  const result = {}, conflicts = [];
  for (const f of files){
    const b = baseT[f], o = oursT[f], t = theirsT[f];
    if (o === t){ if (o !== undefined) result[f] = o; }
    else if (o === b){ if (t !== undefined) result[f] = t; }
    else if (t === b){ if (o !== undefined) result[f] = o; }
    else {
      conflicts.push(f);
      result[f] =
        `<<<<<<< HEAD\n${o !== undefined ? o : ''}=======\n${t !== undefined ? t : ''}>>>>>>> incoming\n`;
    }
  }
  return { result, conflicts };
}

function cmdMerge(args){
  const target = args.find(x => !x.startsWith('-'));
  if (!target){ err('fatal: No commit specified and merge.defaultToUpstream not set.'); return; }
  if (args.includes('--abort')){
    if (!S.mergeState){ err('fatal: There is no merge to abort'); return; }
    S.mergeState = null;
    checkoutTree(headTree());
    out('Merge aborted.');
    return;
  }
  const theirsSha = resolveRev(target);
  if (!theirsSha){ err(`merge: ${target} - not something we can merge`); return; }
  const oursSha = headSha();
  if (!oursSha){ err('fatal: no commits yet — cannot merge'); return; }
  if (theirsSha === oursSha){ out('Already up to date.'); return; }

  const base = mergeBase(oursSha, theirsSha);
  const baseT = base && S.commits[base] ? S.commits[base].tree : {};
  const oursT = S.commits[oursSha].tree;
  const theirsT = S.commits[theirsSha].tree;

  if (base === theirsSha){ out('Already up to date.'); return; }

  if (base === oursSha){
    // fast-forward
    checkoutTree(theirsT);
    if (typeof S.head === 'string') S.branches[S.head] = theirsSha;
    else S.head = { sha: theirsSha };
    reflog(`merge ${target}: Fast-forward`);
    ok(`Fast-forward`);
    dim(` ${Object.keys(theirsT).length} files in tree`);
    return;
  }

  const { result, conflicts } = threeWayMerge(baseT, oursT, theirsT);
  const nonConflict = {};
  for (const f of Object.keys(result)) if (!conflicts.includes(f)) nonConflict[f] = result[f];

  if (conflicts.length){
    S.work = Object.assign({}, S.work, result);
    S.index = nonConflict;
    S.mergeState = { parents: [oursSha, theirsSha], target };
    err(`Auto-merging failed; CONFLICT (content): Merge conflict in ${conflicts.join(', ')}`);
    out('Automatic merge failed; fix conflicts and then commit the result.');
    dim('hint: edit the conflicted file(s), "git add" them, then "git commit -m \\"merge\\""');
    dim('hint: or run "git merge --abort"');
    return;
  }

  const tree = {};
  for (const f of Object.keys(oursT)) tree[f] = oursT[f];
  for (const f of Object.keys(baseT)) if (!(f in theirsT) && !(f in oursT)) delete tree[f];
  for (const f of Object.keys(theirsT)){
    const b = baseT[f], o = oursT[f], t = theirsT[f];
    if (t !== o){
      if (t === undefined) delete tree[f];
      else tree[f] = t;
    }
  }

  const sha = newCommit(tree, `Merge branch '${target}'`, [oursSha, theirsSha]);
  if (typeof S.head === 'string') S.branches[S.head] = sha;
  else S.head = { sha };
  checkoutTree(tree);
  reflog(`merge ${target}: Merge made by the 'ort' strategy.`);
  ok(`Merge made by the 'ort' strategy.`);
  out(` ${Object.keys(tree).length} files in tree`);
}

/* ---------- rebase ---------- */
function cmdRebase(args){
  const target = args.find(x => !x.startsWith('-'));
  if (!target){ err('fatal: no upstream specified'); return; }
  if (args.includes('--abort')){
    out('Rebase aborted (nothing to do in this simulator).');
    return;
  }
  const onto = resolveRev(target);
  if (!onto){ err(`fatal: invalid upstream '${target}'`); return; }
  const ours = headSha();
  if (!ours){ err('fatal: no commits yet'); return; }

  const base = mergeBase(ours, onto);
  if (base === onto){ out('Current branch is up to date.'); return; }

  // collect commits to replay: ours down to (not including) base
  const replay = [];
  let s = ours;
  while (s && s !== base){ replay.push(s); s = S.commits[s].parents[0]; }
  replay.reverse();

  let newBase = onto;
  let tree = Object.assign({}, S.commits[onto].tree);

  for (const sha of replay){
    const c = S.commits[sha];
    const parentT = c.parents[0] && S.commits[c.parents[0]] ? S.commits[c.parents[0]].tree : {};
    // apply this commit's diff onto current tree
    const files = new Set([...Object.keys(parentT), ...Object.keys(c.tree)]);
    for (const f of files){
      const p = parentT[f], n = c.tree[f];
      if (p === n) continue;
      if (n === undefined) delete tree[f];
      else tree[f] = n;
    }
    const newSha = newCommit(tree, c.msg, [newBase]);
    newBase = newSha;
  }

  if (typeof S.head === 'string') S.branches[S.head] = newBase;
  else S.head = { sha: newBase };
  checkoutTree(tree);
  reflog(`rebase onto ${target}`);
  ok(`Successfully rebased and updated refs/heads/${typeof S.head === 'string' ? S.head : 'HEAD'}.`);
  dim(` replayed ${replay.length} commit(s) on top of ${short(onto)}`);
}

/* ---------- cherry-pick ---------- */
function cmdCherryPick(args){
  const ref = args.find(x => !x.startsWith('-'));
  if (!ref){ err('usage: git cherry-pick <commit>'); return; }
  const sha = resolveRev(ref);
  if (!sha){ err(`fatal: bad revision '${ref}'`); return; }
  const c = S.commits[sha];
  const parentT = c.parents[0] && S.commits[c.parents[0]] ? S.commits[c.parents[0]].tree : {};
  const tree = Object.assign({}, headTree());
  const files = new Set([...Object.keys(parentT), ...Object.keys(c.tree)]);
  for (const f of files){
    const p = parentT[f], n = c.tree[f];
    if (p === n) continue;
    if (n === undefined) delete tree[f];
    else tree[f] = n;
  }
  const newSha = newCommit(tree, c.msg, headSha() ? [headSha()] : []);
  if (typeof S.head === 'string') S.branches[S.head] = newSha;
  else S.head = { sha: newSha };
  checkoutTree(tree);
  reflog(`cherry-pick ${short(sha)}`);
  ok(`[${typeof S.head === 'string' ? S.head : 'detached'} ${short(newSha)}] ${c.msg.split('\n')[0]}`);
}

/* ---------- remotes ---------- */
function cmdRemote(args){
  if (!args.length){
    for (const [n, u] of Object.entries(S.remotes)) out(n);
    return;
  }
  if (args[0] === '-v' || args[0] === '--verbose'){
    for (const [n, u] of Object.entries(S.remotes)){
      out(`${n}\t${u} (fetch)`);
      out(`${n}\t${u} (push)`);
    }
    return;
  }
  if (args[0] === 'add'){
    const [name, url] = [args[1], args[2]];
    if (!name || !url){ err('usage: git remote add <name> <url>'); return; }
    if (name in S.remotes){ err(`error: remote ${name} already exists.`); return; }
    S.remotes[name] = url;
    S.remoteRefs[name] = {};
    ok(`remote '${name}' added -> ${url}`);
    return;
  }
  if (args[0] === 'remove' || args[0] === 'rm'){
    const name = args[1];
    if (!(name in S.remotes)){ err(`error: No such remote: '${name}'`); return; }
    delete S.remotes[name];
    delete S.remoteRefs[name];
    ok(`removed remote '${name}'`);
    return;
  }
  err(`error: Unknown subcommand: ${args[0]}`);
}

function cmdFetch(args){
  const name = args.find(x => !x.startsWith('-')) || 'origin';
  if (!(name in S.remotes)){ err(`fatal: '${name}' does not appear to be a git repository`); return; }
  out(`From ${S.remotes[name]}`);
  const refs = S.remoteRefs[name] || {};
  if (!Object.keys(refs).length){ dim(' (no refs to fetch)'); return; }
  for (const [b, sha] of Object.entries(refs)){
    const localTracking = S.remoteRefs[name][b];
    print(`   <span class="dim">${short(sha)}..${short(localTracking)}</span>  ${esc(b)} -> ${esc(name)}/${esc(b)}`);
  }
  ok('fetch complete — remote-tracking refs updated (nothing merged into your branch)');
}

function cmdPull(args){
  const name = args.find(x => !x.startsWith('-')) || 'origin';
  if (!(name in S.remotes)){ err(`fatal: '${name}' does not appear to be a git repository`); return; }
  const branch = typeof S.head === 'string' ? S.head : 'main';
  const remoteSha = (S.remoteRefs[name] || {})[branch];
  if (!remoteSha){ err(`Your configuration specifies to merge with the ref 'refs/heads/${branch}' from the remote, but no such ref was fetched.`); return; }
  out(`From ${S.remotes[name]}`);
  out(` * branch            ${branch}     -> FETCH_HEAD`);
  const local = headSha();
  if (remoteSha === local){ out('Already up to date.'); return; }
  if (isAncestor(remoteSha, local)){ out('Already up to date.'); return; }
  if (isAncestor(local, remoteSha)){
    checkoutTree(S.commits[remoteSha].tree);
    if (typeof S.head === 'string') S.branches[S.head] = remoteSha;
    else S.head = { sha: remoteSha };
    reflog(`pull ${name}/${branch}: Fast-forward`);
    ok('Fast-forward');
    return;
  }
  // diverged -> merge
  const base = mergeBase(local, remoteSha);
  const baseT = base && S.commits[base] ? S.commits[base].tree : {};
  const { result, conflicts } = threeWayMerge(baseT, S.commits[local].tree, S.commits[remoteSha].tree);
  if (conflicts.length){
    S.work = Object.assign({}, S.work, result);
    err(`CONFLICT (content): Merge conflict in ${conflicts.join(', ')}`);
    dim('resolve conflicts, then "git add" and "git commit"');
    return;
  }
  const sha = newCommit(result, `Merge branch '${branch}' of ${name}`, [local, remoteSha]);
  if (typeof S.head === 'string') S.branches[S.head] = sha;
  else S.head = { sha };
  checkoutTree(result);
  reflog(`pull ${name}/${branch}: Merge made by 'ort'`);
  ok(`Merge made by the 'ort' strategy.`);
}

function cmdPush(args){
  const name = args.find(x => !x.startsWith('-')) || 'origin';
  if (!(name in S.remotes)){ err(`fatal: '${name}' does not appear to be a git repository`); return; }
  if (typeof S.head !== 'string'){
    err('fatal: You are not currently on a branch.');
    return;
  }
  const branch = S.head;
  const local = S.branches[branch];
  if (!local){ err('fatal: The current branch has no commits yet.'); return; }
  S.remoteRefs[name] = S.remoteRefs[name] || {};
  const remoteSha = S.remoteRefs[name][branch];

  if (remoteSha && !isAncestor(remoteSha, local)){
    err('To ' + S.remotes[name]);
    err(` ! [rejected]        ${branch} -> ${branch} (non-fast-forward)`);
    dim('error: failed to push some refs to \'' + S.remotes[name] + '\'');
    dim('hint: Updates were rejected because the tip of your current branch is behind');
    dim('hint: its remote counterpart. Try "git pull" first.');
    return;
  }
  if (remoteSha === local){ out('Everything up-to-date'); return; }

  S.remoteRefs[name][branch] = local;
  out(`To ${S.remotes[name]}`);
  print(`   <span class="green">${short(remoteSha || '0000000')}..${short(local)}</span>  ${esc(branch)} -> ${esc(branch)}`);
  ok(`branch '${branch}' pushed to ${name}/${branch}`);
}

/* ---------- undo ---------- */
function cmdRestore(args){
  const staged = args.includes('--staged') || args.includes('-S');
  const files = args.filter(x => !x.startsWith('-'));
  if (!files.length){ err('fatal: you must specify path(s) to restore.'); return; }
  const head = headTree();
  for (const f of files){
    if (staged){
      if (f in head) S.index[f] = head[f];
      else delete S.index[f];
      ok(`restored (staged) '${f}' from HEAD`);
    } else {
      if (f in S.index) S.work[f] = S.index[f];
      else delete S.work[f];
      ok(`restored '${f}' from index`);
    }
  }
}

function cmdReset(args){
  const hard = args.includes('--hard');
  const soft = args.includes('--soft');
  const rest = args.filter(x => !x.startsWith('-'));

  if (hard){
    const target = rest[0] ? resolveRev(rest[0]) : headSha();
    if (rest[0] && !target){ err(`fatal: ambiguous argument '${rest[0]}'`); return; }
    if (target && typeof S.head === 'string') S.branches[S.head] = target;
    else if (target) S.head = { sha: target };
    checkoutTree(target ? S.commits[target].tree : {});
    reflog(`reset --hard ${rest[0] || 'HEAD'}`);
    ok(`HEAD is now at ${short(target) || '(no commits)'} ${target ? S.commits[target].msg.split('\n')[0] : ''}`);
    return;
  }
  if (soft || rest.length){
    const target = rest[0] ? resolveRev(rest[0]) : headSha();
    if (rest[0] && !target){ err(`fatal: ambiguous argument '${rest[0]}'`); return; }
    if (target && typeof S.head === 'string') S.branches[S.head] = target;
    else if (target) S.head = { sha: target };
    if (!soft) checkoutTree(target ? S.commits[target].tree : {});
    reflog(`reset ${soft ? '--soft ' : ''}${rest[0] || 'HEAD'}`);
    ok(`HEAD is now at ${short(target) || '(no commits)'} ${target ? S.commits[target].msg.split('\n')[0] : ''}`);
    return;
  }
  // git reset <file>  -> unstage
  if (!rest.length){ err('usage: git reset <file> | --hard [<commit>] | --soft <commit>'); return; }
  const head = headTree();
  for (const f of rest){
    if (f in head) S.index[f] = head[f];
    else delete S.index[f];
    ok(`unstaged '${f}' (changes kept in working directory)`);
  }
}

function cmdRevert(args){
  const ref = args.find(x => !x.startsWith('-'));
  if (!ref){ err('usage: git revert <commit>'); return; }
  const sha = resolveRev(ref);
  if (!sha){ err(`fatal: bad revision '${ref}'`); return; }
  const c = S.commits[sha];
  const parentT = c.parents[0] && S.commits[c.parents[0]] ? S.commits[c.parents[0]].tree : {};
  const tree = Object.assign({}, headTree());
  const files = new Set([...Object.keys(parentT), ...Object.keys(c.tree)]);
  for (const f of files){
    const p = parentT[f], n = c.tree[f];
    if (p === n) continue;
    if (p === undefined) delete tree[f];
    else tree[f] = p;
  }
  const msg = `Revert "${c.msg.split('\n')[0]}"`;
  const newSha = newCommit(tree, msg, headSha() ? [headSha()] : []);
  if (typeof S.head === 'string') S.branches[S.head] = newSha;
  else S.head = { sha: newSha };
  checkoutTree(tree);
  reflog(`revert: ${msg}`);
  ok(`[${short(newSha)}] ${msg}`);
}

function cmdClean(args){
  const dry = args.includes('-n') || args.includes('--dry-run');
  const head = headTree();
  const untracked = Object.keys(S.work).filter(f => !(f in S.index) && !(f in head));
  if (!untracked.length){ out('nothing to clean'); return; }
  for (const f of untracked){
    if (dry) print(`Would remove ${esc(f)}`);
    else { delete S.work[f]; print(`Removing ${esc(f)}`); }
  }
  if (dry) dim('(dry run — use "git clean -f" to actually delete)');
}

/* ---------- stash ---------- */
function cmdStash(args){
  const sub = args[0] || 'push';
  if (sub === 'push' || sub === 'save'){
    const head = headTree();
    const dirty = [...new Set([...Object.keys(S.work), ...Object.keys(S.index), ...Object.keys(head)])]
      .filter(f => (f in S.work) !== (f in S.index) || S.work[f] !== S.index[f] || (f in S.index) !== (f in head) || S.index[f] !== head[f]);
    if (!dirty.length){ out('No local changes to save'); return; }
    let msg = args.includes('-m') ? args[args.indexOf('-m') + 1] : `WIP on ${branchLabel()}`;
    S.stash.unshift({
      msg, work: Object.assign({}, S.work), index: Object.assign({}, S.index),
      base: headSha(), branch: branchLabel()
    });
    checkoutTree(head);
    ok(`Saved working directory and index state ${msg}`);
    return;
  }
  if (sub === 'list'){
    if (!S.stash.length){ dim('(no stash entries)'); return; }
    S.stash.forEach((s, i) => print(`<span class="yellow">stash@{${i}}</span>: ${esc(s.msg)}`));
    return;
  }
  if (sub === 'pop' || sub === 'apply'){
    if (!S.stash.length){ err('No stash entries found.'); return; }
    const st = S.stash[0];
    for (const f of new Set([...Object.keys(st.work), ...Object.keys(st.index)])){
      if (f in st.work) S.work[f] = st.work[f]; else delete S.work[f];
      if (f in st.index) S.index[f] = st.index[f]; else delete S.index[f];
    }
    if (sub === 'pop'){ S.stash.shift(); ok('Dropped refs/stash@{0}'); }
    ok(`Applied stash@{0} — ${st.msg}`);
    return;
  }
  if (sub === 'drop'){
    if (!S.stash.length){ err('No stash entries found.'); return; }
    S.stash.shift();
    ok('Dropped refs/stash@{0}');
    return;
  }
  if (sub === 'clear'){ S.stash = []; ok('Cleared all stash entries'); return; }
  err(`error: unknown stash subcommand '${sub}'`);
}

/* ---------- inspection ---------- */
function cmdLog(args){
  const oneline = args.includes('--oneline');
  const h = headSha();
  if (!h){ err('fatal: your current branch does not have any commits yet'); return; }
  const shas = [...ancestorsOf(h)].sort((a,b) => S.commits[b].time - S.commits[a].time);
  for (const sha of shas){
    const c = S.commits[sha];
    const labels = [];
    if (typeof S.head === 'string' && S.branches[S.head] === sha) labels.push('HEAD -> ' + S.head);
    else if (typeof S.head !== 'string' && S.head.sha === sha) labels.push('HEAD');
    for (const [b, s2] of Object.entries(S.branches)) if (s2 === sha && b !== S.head) labels.push(b);
    for (const [r, refs] of Object.entries(S.remoteRefs))
      for (const [b, s2] of Object.entries(refs)) if (s2 === sha) labels.push(`${r}/${b}`);

    if (oneline){
      print(`<span class="yellow">${short(sha)}</span> ${esc(c.msg.split('\n')[0])}` +
        (labels.length ? ` <span class="dim">(${esc(labels.join(', '))})</span>` : ''));
    } else {
      print(`<span class="yellow">commit ${sha}</span>` +
        (labels.length ? ` <span class="green">(${esc(labels.join(', '))})</span>` : ''));
      out(`Author: ${c.author} <${c.email}>`);
      out(`Date:   ${fmtDate(c.time)}`);
      blank();
      for (const line of c.msg.split('\n')) print('    ' + esc(line));
      blank();
    }
  }
}

function cmdShow(args){
  const ref = args.find(x => !x.startsWith('-')) || 'HEAD';
  const sha = resolveRev(ref);
  if (!sha){ err(`fatal: ambiguous argument '${ref}'`); return; }
  const c = S.commits[sha];
  print(`<span class="yellow">commit ${sha}</span>`);
  out(`Author: ${c.author} <${c.email}>`);
  out(`Date:   ${fmtDate(c.time)}`);
  blank();
  for (const line of c.msg.split('\n')) print('    ' + esc(line));
  blank();
  const parentT = c.parents[0] && S.commits[c.parents[0]] ? S.commits[c.parents[0]].tree : {};
  const files = new Set([...Object.keys(parentT), ...Object.keys(c.tree)]);
  let any = false;
  for (const f of [...files].sort()){
    const p = parentT[f] !== undefined ? parentT[f] : '';
    const n = c.tree[f] !== undefined ? c.tree[f] : '';
    if (p === n) continue;
    any = true;
    printDiffHeader(f);
    printDiffBody(p, n);
  }
  if (!any) dim('(empty commit)');
}

function cmdBlame(args){
  const f = args.find(x => !x.startsWith('-'));
  if (!f){ err('usage: git blame <file>'); return; }
  const h = headSha();
  if (!h){ err('fatal: no commits yet'); return; }
  if (!(f in S.commits[h].tree)){ err(`fatal: no such path ${f} in HEAD`); return; }

  const chain = [];
  let s = h;
  while (s){ chain.push(s); const c = S.commits[s]; s = c.parents[0]; }
  chain.reverse();

  let attributed = null;
  for (const sha of chain){
    const content = S.commits[sha].tree[f];
    if (content === undefined){ attributed = null; continue; }
    const lines = toLines(content);
    if (attributed === null){
      attributed = lines.map(t => ({ t, sha }));
    } else {
      const ops = lcsDiff(attributed.map(x => x.t), lines);
      const next = []; let oi = 0;
      for (const [op, line] of ops){
        if (op === ' '){ next.push(attributed[oi]); oi++; }
        else if (op === '-'){ oi++; }
        else next.push({ t: line, sha });
      }
      attributed = next;
    }
  }
  attributed.forEach(x => {
    const c = S.commits[x.sha];
    print(`<span class="yellow">${short(x.sha)}</span> <span class="dim">(${esc(c.author.padEnd(14).slice(0,14))} ${fmtDate(c.time).slice(0,10)})</span> ${esc(x.t)}`);
  });
}

function cmdReflog(){
  if (!S.reflog.length){ dim('(no reflog entries)'); return; }
  const list = S.reflog.slice().reverse();
  list.forEach((e, i) => {
    print(`<span class="yellow">${short(newSha())}</span> HEAD@{${i}}: ${esc(e.msg)}`);
  });
}

function buildCheat(){
  const body = $('#cheatBody');
  body.innerHTML = '';
  for (const [group, items] of CHEATS){
    const g = document.createElement('div');
    g.className = 'cheat-group';
    g.innerHTML = `<h4>${esc(group)}</h4>`;
    for (const [cmd, desc] of items){
      const el = document.createElement('div');
      el.className = 'cheat-item';
      el.innerHTML = `<code>${esc(cmd)}</code><span>${esc(desc)}</span>`;
      el.onclick = () => {
        input.value = cmd;
        input.focus();
        $('#cheatModal').classList.remove('show');
      };
      g.appendChild(el);
    }
    body.appendChild(g);
  }
}

/* ============================================================
   WIRING
   ============================================================ */
function runCommand(raw){
  print(`<span class="pgreen">${esc(promptStr())}</span> <span class="cmdtext">${esc(raw)}</span>`);
  const t = raw.trim();
  if (!t) return;
  if (t === 'clear'){ term.innerHTML = ''; return; }
  if (t === 'help' || t === '?'){ printHelp(); return; }
  const argv = tokenize(t);
  if (argv[0] !== 'git'){
    err(`${argv[0]}: command not found`);
    return;
  }
  try { exec(argv.slice(1)); }
  catch (e){ err('fatal: ' + e.message); }
  render();
}

const history = [];
let histIdx = -1;

input.addEventListener('keydown', e => {
  if (e.key === 'Enter'){
    const v = input.value;
    if (v.trim()){ history.push(v); histIdx = history.length; }
    input.value = '';
    runCommand(v);
    render();
  } else if (e.key === 'ArrowUp'){
    e.preventDefault();
    if (histIdx > 0){ histIdx--; input.value = history[histIdx] || ''; }
  } else if (e.key === 'ArrowDown'){
    e.preventDefault();
    if (histIdx < history.length - 1){ histIdx++; input.value = history[histIdx] || ''; }
    else { histIdx = history.length; input.value = ''; }
  }
});

$('#btnTeammate').onclick = () => {
  if (!S.init || !S.remoteRefs.origin){
    err('No remote "origin" configured — run "git clone <url>" or "git remote add origin <url>" first.');
    return;
  }
  const branch = 'main';
  const cur = S.remoteRefs.origin[branch];
  const tree = cur && S.commits[cur] ? Object.assign({}, S.commits[cur].tree) : {};
  tree['notes.md'] = (tree['notes.md'] || '# Team notes\n') + `- teammate edit ${new Date().toLocaleTimeString()}\n`;
  const sha = newCommit(tree, 'Update team notes from teammate', cur ? [cur] : []);
  S.commits[sha].author = 'Teammate';
  S.commits[sha].email = 'teammate@example.com';
  S.remoteRefs.origin[branch] = sha;
  out('☁  Your teammate pushed a new commit to origin/main.');
  dim('   run "git fetch" to inspect it, or "git pull" to bring it in.');
  render();
};
